cmake_minimum_required(VERSION 3.16)
project(rmw_agnocast)
find_package(ament_cmake REQUIRED)
foreach(pkg IN ITEMS std_msgs rosidl_typesupport_fastrtps_cpp rcl_interfaces sensor_msgs agnocastlib rosidl_typesupport_introspection_cpp rclcpp rmw)
  find_package(${pkg} REQUIRED)
endforeach()
add_library(${PROJECT_NAME} SHARED rmw.cpp endpoints.cpp context.cpp gateway_start.cpp filter.cpp events.cpp graph.cpp services.cpp serialization.cpp unsupported.cpp)
target_link_libraries(${PROJECT_NAME} PRIVATE fastcdr ${std_msgs_TARGETS} ${rosidl_typesupport_fastrtps_cpp_TARGETS} ${rcl_interfaces_TARGETS} ${sensor_msgs_TARGETS} ${agnocastlib_TARGETS} ${rosidl_typesupport_introspection_cpp_TARGETS}
  rclcpp::rclcpp rmw::rmw)
register_rmw_implementation("cpp:rosidl_typesupport_introspection_cpp")
install(TARGETS ${PROJECT_NAME} DESTINATION lib)
install(FILES host_cloud_view.hpp DESTINATION include/rmw_agnocast)
ament_export_include_directories(include)
ament_package()
