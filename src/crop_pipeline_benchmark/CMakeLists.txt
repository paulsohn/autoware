cmake_minimum_required(VERSION 3.20)
project(crop_pipeline_benchmark)
find_package(ament_cmake REQUIRED)
foreach(pkg IN ITEMS rclcpp rclcpp_components sensor_msgs tf2_eigen tf2_ros autoware_internal_debug_msgs autoware_internal_msgs shmem_buffer cuda_buffer CUDAToolkit)
  find_package(${pkg} REQUIRED)
endforeach()
set(core "${CMAKE_CURRENT_SOURCE_DIR}/../core/autoware_core/sensing/autoware_crop_box_filter")
set(utils "${CMAKE_CURRENT_SOURCE_DIR}/../core/autoware_utils")
set(patched "${CMAKE_CURRENT_BINARY_DIR}/crop_box")
file(MAKE_DIRECTORY "${patched}")
foreach(name IN ITEMS crop_box_filter.cpp crop_box_filter.hpp crop_box_filter_node.cpp crop_box_filter_node.hpp)
  configure_file("${core}/src/${name}" "${patched}/${name}" COPYONLY)
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS mapped-input.patch)
execute_process(COMMAND patch -p1 -i "${CMAKE_CURRENT_SOURCE_DIR}/mapped-input.patch"
  WORKING_DIRECTORY "${patched}" COMMAND_ERROR_IS_FATAL ANY)
set(WRAPPER_TWO_EDGE "$ENV{WRAPPER_TWO_EDGE}")
if(WRAPPER_TWO_EDGE)
  find_package(autoware_agnocast_wrapper REQUIRED)
  configure_file("${core}/test/test_crop_box_filter_integration.cpp"
    "${patched}/test_crop_box_filter_integration.cpp" COPYONLY)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS two-edge.patch)
  execute_process(COMMAND patch -p1 -i "${CMAKE_CURRENT_SOURCE_DIR}/two-edge.patch"
    WORKING_DIRECTORY "${patched}" COMMAND_ERROR_IS_FATAL ANY)
endif()
add_library(crop_component SHARED "${patched}/crop_box_filter.cpp" "${patched}/crop_box_filter_node.cpp")
target_include_directories(crop_component PUBLIC "${patched}")
foreach(module IN ITEMS debug tf system)
  target_include_directories(crop_component PUBLIC "${utils}/autoware_utils_${module}/include")
endforeach()
target_link_libraries(crop_component PUBLIC shmem_buffer::shmem_buffer tf2_eigen::tf2_eigen tf2_ros::tf2_ros
  rclcpp_components::component ${autoware_internal_debug_msgs_TARGETS} ${autoware_internal_msgs_TARGETS} ${sensor_msgs_TARGETS})
if(WRAPPER_TWO_EDGE)
  target_include_directories(crop_component PUBLIC ${autoware_agnocast_wrapper_INCLUDE_DIRS})
  target_link_libraries(crop_component PUBLIC ${autoware_agnocast_wrapper_LIBRARIES}
    cuda_buffer::cuda_buffer CUDA::cudart)
endif()
rclcpp_components_register_node(crop_component PLUGIN "autoware::crop_box_filter::CropBoxFilterNode" EXECUTABLE crop)
add_executable(endpoint endpoint.cpp)
target_link_libraries(endpoint PRIVATE rclcpp::rclcpp ${sensor_msgs_TARGETS} shmem_buffer::shmem_buffer cuda_buffer::cuda_buffer CUDA::cudart)
install(TARGETS crop_component DESTINATION lib)
install(TARGETS endpoint DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  foreach(suite IN ITEMS test_crop_box_filter test_crop_box_filter_integration)
    set(test_source "${core}/test/${suite}.cpp")
    if(WRAPPER_TWO_EDGE AND suite STREQUAL "test_crop_box_filter_integration")
      set(test_source "${patched}/${suite}.cpp")
    endif()
    ament_add_gtest(${suite} "${test_source}")
    target_link_libraries(${suite} crop_component)
  endforeach()
endif()
ament_package()
