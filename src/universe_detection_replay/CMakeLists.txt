cmake_minimum_required(VERSION 3.16)
project(universe_detection_replay)
find_package(ament_cmake REQUIRED)
set(packages rclcpp_components rclcpp sensor_msgs pcl_conversions tier4_perception_msgs
  autoware_internal_debug_msgs)
foreach(pkg IN LISTS packages)
  find_package(${pkg} REQUIRED)
endforeach()
find_package(PCL REQUIRED COMPONENTS segmentation)
set(node "${CMAKE_CURRENT_SOURCE_DIR}/../universe/autoware_universe/perception/autoware_euclidean_cluster")
add_library(cluster SHARED "${node}/src/euclidean_cluster_node.cpp"
  "${node}/lib/euclidean_cluster.cpp" "${node}/lib/utils.cpp")
target_include_directories(cluster PRIVATE "${node}/include" ${PCL_INCLUDE_DIRS})
foreach(pkg IN ITEMS autoware_utils autoware_utils_debug autoware_utils_system)
  target_include_directories(cluster PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../core/autoware_utils/${pkg}/include")
endforeach()
foreach(pkg IN LISTS packages)
  target_link_libraries(cluster ${${pkg}_TARGETS})
  target_include_directories(cluster SYSTEM PRIVATE ${${pkg}_INCLUDE_DIRS})
endforeach()
target_link_libraries(cluster ${PCL_LIBRARIES})
rclcpp_components_register_node(cluster
  PLUGIN "autoware::euclidean_cluster::EuclideanClusterNode" EXECUTABLE replay_cluster)
install(TARGETS cluster LIBRARY DESTINATION lib)
ament_package()
