cmake_minimum_required(VERSION 3.16)
project(universe_crop_benchmark)
find_package(ament_cmake REQUIRED)
set(deps rclcpp rclcpp_components agnocastlib autoware_agnocast_wrapper managed_transform_buffer
  sensor_msgs geometry_msgs pcl_ros pcl_conversions pcl_msgs point_cloud_msg_wrapper
  autoware_internal_debug_msgs autoware_internal_msgs)
foreach(pkg IN LISTS deps)
  find_package(${pkg} REQUIRED)
endforeach()
set(aw "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(pre "${aw}/universe/autoware_universe/sensing/autoware_pointcloud_preprocessor")
option(UNIVERSE_BUFFER_MATRIX "Build explicit Lyrical buffer cases" OFF)
set(crop_src "${pre}")
if(UNIVERSE_BUFFER_MATRIX)
  find_package(shmem_buffer REQUIRED)
  find_package(cuda_buffer REQUIRED)
  find_package(CUDAToolkit REQUIRED)
  set(wrapper_include "${aw}/core/autoware_core/common/autoware_agnocast_wrapper/include")
  set(publisher_header autoware/agnocast_wrapper/publisher.hpp)
  set(wrapper_copy "${CMAKE_CURRENT_BINARY_DIR}/loan_wrapper")
  configure_file("${wrapper_include}/${publisher_header}" "${wrapper_copy}/${publisher_header}" COPYONLY)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS publisher-loan.patch)
  execute_process(COMMAND patch -p1 -i "${CMAKE_CURRENT_SOURCE_DIR}/publisher-loan.patch"
    WORKING_DIRECTORY "${wrapper_copy}" COMMAND_ERROR_IS_FATAL ANY)
  set(crop_src "${CMAKE_CURRENT_BINARY_DIR}/buffer_crop")
  foreach(file IN ITEMS include/autoware/pointcloud_preprocessor/filter.hpp
      src/filter.cpp src/crop_box_filter/crop_box_filter_node.cpp)
    configure_file("${pre}/${file}" "${crop_src}/${file}" COPYONLY)
  endforeach()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS buffer-transport.patch)
  execute_process(COMMAND patch -p1 -i "${CMAKE_CURRENT_SOURCE_DIR}/buffer-transport.patch"
    WORKING_DIRECTORY "${crop_src}" COMMAND_ERROR_IS_FATAL ANY)
endif()
set(crop_impl "${crop_src}/src/crop_box_filter/crop_box_filter_node.cpp")
set(filter_impl "${crop_src}/src/filter.cpp")
if(NOT UNIVERSE_BUFFER_MATRIX)
  configure_file("${filter_impl}" "${CMAKE_CURRENT_BINARY_DIR}/filter.cpp" COPYONLY)
  set(filter_impl "${CMAKE_CURRENT_BINARY_DIR}/filter.cpp")
  configure_file("${crop_impl}" "${CMAKE_CURRENT_BINARY_DIR}/crop.cpp" COPYONLY)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS region-stall.patch)
  execute_process(COMMAND patch -p1 -i "${CMAKE_CURRENT_SOURCE_DIR}/region-stall.patch"
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}" COMMAND_ERROR_IS_FATAL ANY)
  set(crop_impl "${CMAKE_CURRENT_BINARY_DIR}/crop.cpp")
endif()
add_library(universe_crop SHARED "${filter_impl}" "${pre}/src/utility/memory.cpp"
  "${crop_impl}")
target_include_directories(universe_crop PRIVATE "${crop_src}/include" "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(universe_crop PRIVATE "${pre}/include"
  "${aw}/core/autoware_core/common/autoware_point_types/include")
foreach(pkg IN ITEMS autoware_utils autoware_utils_debug autoware_utils_diagnostics autoware_utils_system)
  target_include_directories(universe_crop PRIVATE "${aw}/core/autoware_utils/${pkg}/include")
endforeach()
foreach(pkg IN LISTS deps)
  target_include_directories(universe_crop SYSTEM PRIVATE ${${pkg}_INCLUDE_DIRS})
  if(${pkg}_TARGETS)
    target_link_libraries(universe_crop ${${pkg}_TARGETS})
  else()
    target_link_libraries(universe_crop ${${pkg}_LIBRARIES})
  endif()
endforeach()
autoware_agnocast_wrapper_register_node(universe_crop
  PLUGIN "autoware::pointcloud_preprocessor::CropBoxFilterComponent" EXECUTABLE crop
  AGNOCAST_EXECUTOR AgnocastOnlyCallbackIsolatedExecutor ROS2_EXECUTOR SingleThreadedExecutor)
if("$ENV{ENABLE_AGNOCAST}" STREQUAL "1")
  add_executable(native_crop_container native_crop_container.cpp)
  target_include_directories(native_crop_container PRIVATE ${autoware_agnocast_wrapper_INCLUDE_DIRS})
  target_link_libraries(native_crop_container PRIVATE universe_crop ${agnocastlib_TARGETS} ${rclcpp_components_TARGETS})
  install(TARGETS native_crop_container DESTINATION lib/${PROJECT_NAME})
endif()
add_library(universe_endpoint SHARED endpoint.cpp)
target_link_libraries(universe_endpoint ${rclcpp_TARGETS} ${rclcpp_components_TARGETS}
  ${sensor_msgs_TARGETS} ${agnocastlib_TARGETS} ${autoware_agnocast_wrapper_LIBRARIES})
target_include_directories(universe_endpoint PRIVATE ${autoware_agnocast_wrapper_INCLUDE_DIRS})
autoware_agnocast_wrapper_register_node(universe_endpoint
  PLUGIN "benchmark::Endpoint" EXECUTABLE endpoint
  AGNOCAST_EXECUTOR AgnocastOnlySingleThreadedExecutor ROS2_EXECUTOR SingleThreadedExecutor)
if(NOT UNIVERSE_BUFFER_MATRIX AND NOT "$ENV{ENABLE_AGNOCAST}" STREQUAL "1")
  rclcpp_components_register_node(universe_crop
    PLUGIN "autoware::pointcloud_preprocessor::CropBoxFilterComponent" EXECUTABLE crop_two
    EXECUTOR crop_benchmark::TwoWorkerExecutor RESOURCE_INDEX universe_crop_benchmark_two)
  target_compile_options(crop_two PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/two_worker_executor.hpp")
endif()
if(UNIVERSE_BUFFER_MATRIX OR NOT "$ENV{ENABLE_AGNOCAST}" STREQUAL "1")
  # ROS-only crop process using the CIE shipped with our pinned Agnocast.
  rclcpp_components_register_node(universe_crop
    PLUGIN "autoware::pointcloud_preprocessor::CropBoxFilterComponent" EXECUTABLE crop_cie
    EXECUTOR agnocast::CallbackIsolatedAgnocastExecutor RESOURCE_INDEX universe_crop_benchmark_cie)
  target_link_libraries(crop_cie ${agnocastlib_TARGETS})
  target_compile_options(crop_cie PRIVATE -include agnocast/agnocast_callback_isolated_executor.hpp)
endif()
if(UNIVERSE_BUFFER_MATRIX)
  foreach(target IN ITEMS universe_crop universe_endpoint)
    target_include_directories(${target} BEFORE PRIVATE "${wrapper_copy}")
    target_link_libraries(${target} shmem_buffer::shmem_buffer cuda_buffer::cuda_buffer CUDA::cudart)
    target_compile_features(${target} PRIVATE cxx_std_20)
  endforeach()
  target_compile_definitions(universe_endpoint PRIVATE UNIVERSE_BUFFER_MATRIX)
endif()
install(TARGETS universe_crop universe_endpoint DESTINATION lib)
ament_package()
