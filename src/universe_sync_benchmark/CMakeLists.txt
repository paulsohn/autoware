cmake_minimum_required(VERSION 3.16)
project(universe_sync_benchmark)
find_package(ament_cmake REQUIRED)
set(deps rclcpp_components rclcpp sensor_msgs nav_msgs geometry_msgs autoware_internal_debug_msgs
  autoware_sensing_msgs diagnostic_updater managed_transform_buffer point_cloud_msg_wrapper
  message_filters pcl_conversions pcl_ros)
set(aw "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(pre "${aw}/universe/autoware_universe/sensing/autoware_pointcloud_preprocessor")
set(sync "${pre}/src/concatenate_data")
option(AUTOWARE_ENABLE_AGNOCAST_HOST_VIEW "Build the experimental Lyrical host-view input" ON)
option(SYNC_NATIVE "Build native Agnocast cloud endpoints and CIE" OFF)
if(SYNC_NATIVE AND NOT AUTOWARE_ENABLE_AGNOCAST_HOST_VIEW)
  message(FATAL_ERROR "Native comparison requires the matched input converter")
endif()
set(input_sources "${sync}")
if(AUTOWARE_ENABLE_AGNOCAST_HOST_VIEW)
  set(input_sources "${CMAKE_CURRENT_BINARY_DIR}/loan-input")
  foreach(file IN ITEMS concatenate_and_time_sync_node.cpp combine_cloud_handler.cpp)
    configure_file("${sync}/${file}" "${input_sources}/${file}" COPYONLY)
  endforeach()
  set(node_impl autoware/pointcloud_preprocessor/concatenate_data/concatenate_and_time_sync_node.ipp)
  configure_file("${pre}/include/${node_impl}" "${input_sources}/${node_impl}" COPYONLY)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS loan-input.patch loan-output.patch)
  execute_process(COMMAND patch -p1 -i "${CMAKE_CURRENT_SOURCE_DIR}/loan-input.patch"
    WORKING_DIRECTORY "${input_sources}" COMMAND_ERROR_IS_FATAL ANY)
  if(SYNC_NATIVE)
    set(node_header autoware/pointcloud_preprocessor/concatenate_data/concatenate_and_time_sync_node.hpp)
    configure_file("${pre}/include/${node_header}" "${input_sources}/${node_header}" COPYONLY)
    set(traits autoware/pointcloud_preprocessor/concatenate_data/traits.hpp)
    configure_file("${pre}/include/${traits}" "${input_sources}/${traits}" COPYONLY)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS native.patch)
    set(output_patch native.patch)
  else()
    set(output_patch loan-output.patch)
  endif()
  execute_process(COMMAND patch -p1 -i "${CMAKE_CURRENT_SOURCE_DIR}/${output_patch}"
    WORKING_DIRECTORY "${input_sources}" COMMAND_ERROR_IS_FATAL ANY)
  if(NOT SYNC_NATIVE)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS stage-trace.patch)
    execute_process(COMMAND patch -p1 -i "${CMAKE_CURRENT_SOURCE_DIR}/stage-trace.patch"
      WORKING_DIRECTORY "${input_sources}" COMMAND_ERROR_IS_FATAL ANY)
  endif()
endif()
add_library(universe_sync SHARED
  "${input_sources}/concatenate_and_time_sync_node.cpp" "${sync}/cloud_collector.cpp"
  "${sync}/collector_matcher.cpp" "${sync}/matching_policy.cpp"
  "${input_sources}/combine_cloud_handler.cpp" "${sync}/combine_cloud_handler_base.cpp"
  "${sync}/concatenation_diagnostics.cpp" "${pre}/src/utility/memory.cpp")
if(AUTOWARE_ENABLE_AGNOCAST_HOST_VIEW)
  find_package(rmw_agnocast REQUIRED)
  find_package(agnocastlib REQUIRED)
  target_link_libraries(universe_sync ${agnocastlib_TARGETS})
  target_include_directories(universe_sync BEFORE PRIVATE "${input_sources}")
  target_include_directories(universe_sync PRIVATE "${sync}" "${CMAKE_CURRENT_SOURCE_DIR}" ${rmw_agnocast_INCLUDE_DIRS})
  target_compile_features(universe_sync PRIVATE cxx_std_20)
endif()
target_include_directories(universe_sync PRIVATE "${pre}/include"
  "${aw}/core/autoware_core/common/autoware_point_types/include")
foreach(pkg IN ITEMS autoware_utils autoware_utils_debug autoware_utils_diagnostics autoware_utils_system)
  target_include_directories(universe_sync PRIVATE "${aw}/core/autoware_utils/${pkg}/include")
endforeach()
foreach(pkg IN LISTS deps)
  find_package(${pkg} REQUIRED)
  target_include_directories(universe_sync SYSTEM PRIVATE ${${pkg}_INCLUDE_DIRS})
  if(${pkg}_TARGETS)
    target_link_libraries(universe_sync ${${pkg}_TARGETS})
  else()
    target_link_libraries(universe_sync ${${pkg}_LIBRARIES})
  endif()
endforeach()
if(SYNC_NATIVE)
  set(executor EXECUTOR native_sync::Executor)
endif()
rclcpp_components_register_node(universe_sync ${executor}
  PLUGIN "autoware::pointcloud_preprocessor::PointCloudConcatenateDataSynchronizerComponent"
  EXECUTABLE concatenate_and_time_sync_node)
install(TARGETS universe_sync DESTINATION lib)
add_executable(sync_fixture fixture.cpp)
target_link_libraries(sync_fixture ${rclcpp_TARGETS} ${sensor_msgs_TARGETS} ${autoware_sensing_msgs_TARGETS})
install(TARGETS sync_fixture DESTINATION lib/${PROJECT_NAME})
if(SYNC_NATIVE)
  target_include_directories(universe_sync PRIVATE "${pre}/include/autoware/pointcloud_preprocessor/concatenate_data")
  target_compile_options(concatenate_and_time_sync_node PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/native_executor.hpp")
  target_link_libraries(concatenate_and_time_sync_node ${agnocastlib_TARGETS})
  target_compile_definitions(sync_fixture PRIVATE SYNC_NATIVE)
  target_include_directories(sync_fixture PRIVATE ${rmw_agnocast_INCLUDE_DIRS})
  target_link_libraries(sync_fixture ${agnocastlib_TARGETS})
endif()
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(FILES "${pre}/config/crop_box_filter_node.param.yaml"
  "${pre}/config/concatenate_and_time_sync_node.param.yaml" DESTINATION share/${PROJECT_NAME}/config)
ament_package()
