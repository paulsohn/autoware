cmake_minimum_required(VERSION 3.16)
project(agnocast_rmw_loan_probe)
find_package(ament_cmake REQUIRED)
foreach(pkg IN ITEMS sensor_msgs nav_msgs geometry_msgs agnocastlib autoware_planning_msgs rclcpp rmw)
  find_package(${pkg} REQUIRED)
endforeach()
add_executable(loan_probe probe.cpp loan_hooks.cpp)
set_target_properties(loan_probe PROPERTIES ENABLE_EXPORTS ON)
target_link_libraries(loan_probe ${agnocastlib_TARGETS} ${autoware_planning_msgs_TARGETS}
  rclcpp::rclcpp rmw::rmw ${CMAKE_DL_LIBS})
install(TARGETS loan_probe DESTINATION lib/${PROJECT_NAME})
add_executable(loadable_probe loadable_probe.cpp)
target_link_libraries(loadable_probe ${sensor_msgs_TARGETS} ${nav_msgs_TARGETS} ${geometry_msgs_TARGETS} ${autoware_planning_msgs_TARGETS} rclcpp::rclcpp rmw::rmw)
install(TARGETS loadable_probe DESTINATION lib/${PROJECT_NAME})
add_executable(type_fallback_peer type_fallback_peer.cpp)
target_link_libraries(type_fallback_peer PRIVATE ${geometry_msgs_TARGETS} ${sensor_msgs_TARGETS} ${nav_msgs_TARGETS} rclcpp::rclcpp)
install(TARGETS type_fallback_peer DESTINATION lib/${PROJECT_NAME})
# The historical manual_bridge and route_bridge used native and DDS endpoints
# in one mixed RMW. Neither is a standalone network bridge.
add_executable(path_benchmark path_benchmark.cpp)
target_link_libraries(path_benchmark ${agnocastlib_TARGETS} ${nav_msgs_TARGETS} rclcpp::rclcpp rmw::rmw)
target_link_options(path_benchmark PRIVATE
  -Wl,--export-dynamic-symbol=path_probe_wake -Wl,--export-dynamic-symbol=path_probe_loan)
install(TARGETS path_benchmark DESTINATION lib/${PROJECT_NAME})
find_package(rclcpp_components REQUIRED)
find_package(class_loader REQUIRED)
add_executable(crop_wait_runner crop_wait_runner.cpp)
target_link_libraries(crop_wait_runner rclcpp_components::component_manager
  class_loader::class_loader rclcpp::rclcpp ${CMAKE_DL_LIBS})
install(TARGETS crop_wait_runner DESTINATION lib/${PROJECT_NAME})
add_executable(lifecycle_container lifecycle_container.cpp)
target_link_libraries(lifecycle_container rclcpp_components::component_manager rclcpp::rclcpp ${CMAKE_DL_LIBS})
install(TARGETS lifecycle_container DESTINATION lib/${PROJECT_NAME})
foreach(probe IN ITEMS receive_rcl_trace receive_executor_trace)
  add_library(${probe} SHARED receive_rcl_trace.cpp)
  target_link_libraries(${probe} PRIVATE rclcpp::rclcpp ${CMAKE_DL_LIBS})
  install(TARGETS ${probe} DESTINATION lib)
endforeach()
target_compile_definitions(receive_executor_trace PRIVATE BENCH_EXECUTOR_SPLIT)
# The standalone interposition probe also builds before the experimental RMW exists.
find_package(rmw_agnocast QUIET)
if(rmw_agnocast_FOUND)
  add_executable(mixed_cloud mixed_cloud.cpp)
  target_include_directories(mixed_cloud PRIVATE ${rmw_agnocast_INCLUDE_DIRS})
  target_link_libraries(mixed_cloud ${sensor_msgs_TARGETS} ${agnocastlib_TARGETS} rclcpp::rclcpp)
  install(TARGETS mixed_cloud DESTINATION lib/${PROJECT_NAME})
endif()
ament_package()
