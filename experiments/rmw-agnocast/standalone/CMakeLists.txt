cmake_minimum_required(VERSION 3.16)
project(standalone_topic_check)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(topic_check topic_check.cpp)
target_link_libraries(topic_check PRIVATE rclcpp::rclcpp std_msgs::std_msgs)

find_package(rcl_interfaces REQUIRED)
add_executable(rosout_check rosout_check.cpp)
target_link_libraries(rosout_check PRIVATE rclcpp::rclcpp ${rcl_interfaces_TARGETS})
find_package(std_srvs REQUIRED)
add_executable(service_check service_check.cpp)
target_link_libraries(service_check PRIVATE rclcpp::rclcpp ${std_srvs_TARGETS})
add_executable(api_check api_check.cpp)
target_link_libraries(api_check PRIVATE rclcpp::rclcpp std_msgs::std_msgs)
